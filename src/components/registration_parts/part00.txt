async function findPlayerAccountByPhone(phone: string) {
  const candidates = phoneLookupCandidates(phone);
  for (const candidate of candidates) {
    const { data } = await supabase
      .from('player_accounts')
      .select('*')
      .eq('phone_number', candidate)
      .maybeSingle();
    if (data) return data as PlayerAccount;
  }

  // Fallback: national digits (handles +34 vs +33 for same mobile body)
  const key = candidates
    .map((c) => c.replace(/\D/g, ''))
    .sort((a, b) => b.length - a.length)[0];
  const last9 = key?.slice(-9);
  if (last9 && last9.length >= 9) {
    const { data: rows } = await supabase
      .from('player_accounts')
      .select('*')
      .ilike('phone_number', `%${last9}`)
      .limit(20);
    const national = normalizePhoneKey(phone);
    const matches = (rows || []).filter((r: any) =>
      phonesEqual(r.phone_number, phone) || normalizePhoneKey(r.phone_number) === national
    );
    if (matches.length === 1) return matches[0] as PlayerAccount;
    if (matches.length > 1) {
      const exact = matches.find((r: any) => candidates.includes(r.phone_number) || phonesEqual(r.phone_number, phone));
      return (exact || matches[0]) as PlayerAccount;
    }
  }
  return null;
}
